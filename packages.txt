# packages.txt — the programs on the MyOS live disk.
#
# Unmodified 32-bit x86 programs from Alpine Linux 3.20 (main + community).
# Only the top-level packages are listed: tools/fetch-live.py adds the
# libraries each one needs (and, like apk, the "install_if" extras such as
# man pages), and pins every exact version and SHA-256 in
# live/packages.lock. `make live-root` builds build/live/rootfs.tar.
#
# A line "[name]" starts a group (the size report adds up each group).
# A line "!name" forbids a package: the build stops if anything needs it.
#
# The live disk is a normal Linux userland, the way Debian, Fedora or Void
# Linux ship one: GNU coreutils, util-linux, procps, a real init and login.
# No BusyBox. BusyBox stays where it is good — in the kernel's test
# initramfs (tools/fetch-alpine.sh, `make test`), where one small static
# program gives the self-tests ~300 commands. (Alpine itself uses BusyBox
# for all of this even on full installs; we take Alpine's packages but not
# that choice.)
!busybox

[init]
# Process 1 is runit, set up the way Void Linux does it: /etc/runit/1 boots
# the machine, /etc/runit/2 supervises the services in /etc/sv (one small
# "run" script each), /etc/runit/3 shuts down. agetty puts a login prompt
# on each terminal; util-linux's login checks the password (PAM, the
# /etc/shadow file) and starts the user's shell as that user. shadow gives
# useradd, usermod, passwd; doas lets the wheel group act as root (57 KB,
# sudo is 1.7 MB). xdm is the graphical login screen: small, no D-Bus.
runit
agetty
util-linux-login
shadow
doas
xdm
