# packages.txt — the programs on the MyOS live disk.
#
# Unmodified 32-bit x86 programs from Alpine Linux 3.20 (main + community).
# Only the top-level packages are listed: tools/fetch-live.py adds the
# libraries each one needs (and, like apk, the "install_if" extras such as
# man pages), and pins every exact version and SHA-256 in
# live/packages.lock. `make live-root` builds build/live/rootfs.tar.
#
# A line "[name]" starts a group (the size report adds up each group).
# A line "!name" forbids a package: the build stops if anything needs it.
#
# The live disk is a normal Linux userland, the way Debian, Fedora or Void
# Linux ship one: GNU coreutils, util-linux, procps, a real init and login.
# No BusyBox. BusyBox stays where it is good — in the kernel's test
# initramfs (tools/fetch-alpine.sh, `make test`), where one small static
# program gives the self-tests ~300 commands. (Alpine itself uses BusyBox
# for all of this even on full installs; we take Alpine's packages but not
# that choice.)
!busybox

[init]
# Process 1 is runit, set up the way Void Linux does it: /etc/runit/1 boots
# the machine, /etc/runit/2 supervises the services in /etc/sv (one small
# "run" script each), /etc/runit/3 shuts down. agetty puts a login prompt
# on each terminal; util-linux's login checks the password (PAM, the
# /etc/shadow file) and starts the user's shell as that user. shadow gives
# useradd, usermod, passwd; doas lets the wheel group act as root (57 KB,
# sudo is 1.7 MB). The graphical login screen is MyOS's own (myos-login,
# live/myos-login/), built from the [dev] group's headers.
# sshd is supervised like agetty (live/overlay/etc/sv/sshd). The client is
# also in [dev]. Host keys are made on first boot. Incoming TCP still needs
# the kernel's listen/accept (docs/gaps.md).
# socklog (31 KB) is the system log, the runit way: it reads /dev/log, where
# login, PAM and the daemons send their messages (syslog(3)), and runit's own
# svlogd writes them to /var/log/socklog, rotating its files so the log can't
# fill the disk (live/overlay/etc/sv/socklog-unix).
runit
socklog
agetty
util-linux-login
shadow
doas
openssh-server

[fonts]
# Chinese, Japanese and Korean text: Noto Sans CJK and Noto Serif CJK
# (Google and Adobe's Source Han), regular and bold. DejaVu has no CJK
# letters at all, so without these the installer's first page showed 中文
# and 日本語 as boxes. One font file per style holds the Simplified Chinese,
# Traditional Chinese, Japanese and Korean shapes of each character;
# fontconfig picks by the text's language. 93 MB, so not in the RAM login
# image (LOGIN_GROUPS). On an installed system it comes with the installer's
# "Also install fonts and input for Chinese" (docs/INSTALLER-PLAN.md).
font-noto-cjk
# Open Sans: the Golden Hour headings (the mockups say Trebuchet MS, which
# isn't free; Open Sans is the closest free humanist sans). 3 MB. Body text
# stays DejaVu Sans, the mockups' own stand-in for Verdana.
font-opensans
